**第 3 项脚本**（每张 md 表格 = 1 表头行 + 1 分隔行 + N 数据行，故 `真数据 = 总行数 - 2`）：

```bash
"$PY" -X utf8 -c "
p='备课思路/未央化学竞赛课程计划（初二至高二）.md'
t=open(p,encoding='utf-8',newline='').read()
blocks=[];cur=[]
for l in t.split(chr(10)):
    if l.startswith('|'): cur.append(l)
    else:
        if cur: blocks.append(cur); cur=[]
if cur: blocks.append(cur)
tot=0
for i,b in enumerate(blocks,1):
    sep=[x for x in b if set(x.strip().strip('|'))<=set(':-| ')]  # ⚠️ 集合必须含 '|'——漏写会把 9 张表的分隔行全判 0、数据行虚高（330→348），2026-09-15 核验会话实测踩过
    dat=len(b)-len(sep)-1   # 再扣 1 行表头
    tot+=dat
    print('  表%d: 总行=%d 分隔=%d 数据=%d'%(i,len(b),len(sep),dat))
print('  合计数据行 =',tot)
"
```
