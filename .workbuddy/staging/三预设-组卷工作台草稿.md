## 预设 1 · 习题集卷（基础+巩固档）

```dataviewjs
// ══════════ 预设 1 · 习题集卷 ══════════
// 书+submodule 二级筛选 → 难度升序 15 题 → 答案随卷（卷尾附答案跳转）。
// 用法：先看下方「可选书名 / 子模块」清单，把想练的填进 BOOK / SUB，改 SEED 换一版。
const LEVELS = ["基础", "巩固"];   // 习题集档 = 基础 + 巩固（三模块视图，勿改）
const BOOK   = "";                 // 一级：书名（srcKey 归一后），"" 不限
const SUB    = "";                 // 二级：submodule 字段值，"" 不限
const TOTAL  = 15;
const SEED   = 20260902;
const EXCLUDE_USED = true;

// ── 工具函数（dataviewjs 块间不共享作用域，每块重定义）──
const hasUsed = u => u != null && (Array.isArray(u) ? u.length > 0 : true);
const diffNum = d => { const m = String(d ?? "").match(/\d+/); return m ? Number(m[0]) : NaN; };
function srcKey(s) {
  let k = String(s || "").trim();
  k = k.replace(/[（(]忠实转录[)）]/g, "");
  k = k.replace(/[·•]\s*第\s*\d+\s*[讲章][\s\S]*$/, "");
  k = k.replace(/第\s*[\d一二三四五六七八九十]+\s*[章节讲篇][\s\S]*$/, "");
  k = k.replace(/[·•]\s*第\s*[\d一二三四五六七八九十]+\s*[分册卷][\s\S]*$/, "");
  k = k.replace(/-\s*\d{1,2}\s*$/, "");
  k = k.replace(/[\s·•、,，\-—]+$/, "");
  return k.trim() || "(未知来源)";
}
function mkRand(seed) {
  let a = seed >>> 0;
  return function () {
    a = (a + 0x6D2B79F5) >>> 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
const rand = mkRand(SEED);

const ALL = dv.pages('"04-题库" or "05-真题库"').where(p => {
  if (p.type !== "题目" && p.type !== "真题") return false;
  if (p.status === "deprecated") return false;
  if (!LEVELS.includes(p.teaching_level)) return false;
  if (EXCLUDE_USED && hasUsed(p.used_in)) return false;
  return !Number.isNaN(diffNum(p.difficulty));
}).array();

// 二级筛选
const pool = ALL.filter(p =>
  (!BOOK || srcKey(p.source) === BOOK) &&
  (!SUB || String(p.submodule ?? "") === SUB));

// ── 可选值清单（填 BOOK / SUB 的参考）──
const bookCnt = new Map(), subCnt = new Map();
for (const p of ALL) {
  const b = srcKey(p.source);
  bookCnt.set(b, (bookCnt.get(b) || 0) + 1);
  const s = String(p.submodule ?? "").trim();
  if (s) subCnt.set(s, (subCnt.get(s) || 0) + 1);
}
dv.header(4, `📚 习题集卷候选：${pool.length} 题（档内总量 ${ALL.length}）`);
dv.paragraph("**可填书名**（档内题量 top 12）：" +
  [...bookCnt.entries()].sort((a, b) => b[1] - a[1]).slice(0, 12)
    .map(([k, v]) => `${k.slice(0, 20)}×${v}`).join(" ｜ "));
dv.paragraph("**可填子模块**（submodule，档内题量 top 12）：" +
  [...subCnt.entries()].sort((a, b) => b[1] - a[1]).slice(0, 12)
    .map(([k, v]) => `${k}×${v}`).join(" ｜ "));

// ── 成卷：难度升序（同难度按种子稳定打乱）──
pool.sort((a, b) => diffNum(a.difficulty) - diffNum(b.difficulty) || rand() - 0.5);
const picked = pool.slice(0, Math.min(TOTAL, pool.length));

dv.table(["#", "题目", "难度", "题型", "模块", "submodule", "来源"],
  picked.map((p, i) => [
    i + 1, p.file.link, p.difficulty,
    (Array.isArray(p.question_type) ? p.question_type.join("/") : p.question_type) || "—",
    p.subject_module, p.submodule || "—", srcKey(p.source).slice(0, 18)
  ]));
dv.paragraph("**参考答案**（随卷）：\n" +
  picked.map((p, i) => `${i + 1}. ${p.file.link} → 见题目笔记「参考答案」`).join("\n"));
dv.paragraph(`**复制清单**（粘到试卷文件后跑 mark_used.py 回填）\n：` +
  picked.map((p, i) => `${i + 1}. ${p.file.link}`).join("\n"));
```

## 预设 2 · 习题书卷（拓展档）

```dataviewjs
// ══════════ 预设 2 · 习题书卷 ══════════
// KP 精确命中 → 难度升序 → 答案卷尾。
// KP 名必须与题目 knowledge_points 的实际写法完全一致（见下方候选清单）。
const KPS  = ["化学平衡"];         // 可多个：["配位化合物", "晶体场理论"]
const TOTAL = 15;
const SEED  = 20260902;
const EXCLUDE_USED = true;

// ── 工具函数（dataviewjs 块间不共享作用域，每块重定义）──
const hasUsed = u => u != null && (Array.isArray(u) ? u.length > 0 : true);
const diffNum = d => { const m = String(d ?? "").match(/\d+/); return m ? Number(m[0]) : NaN; };
// knowledge_points 元素是 wikilink → dataview 解析成 Link，取 path 末段去 .md 再比对
const kpNames = p => {
  const arr = Array.isArray(p.knowledge_points) ? p.knowledge_points
            : (p.knowledge_points ? [p.knowledge_points] : []);
  return arr.map(k => String(k?.path ?? k ?? "").split("/").pop().replace(/\.md$/, ""))
            .filter(Boolean);
};
// FNV-1a：种子稳定 tie-break（与 gen_weak_drill.py 的 md5 同目的：同参数=同卷）
const h = s => { let x = 2166136261; for (const c of String(s)) { x ^= c.charCodeAt(0); x = Math.imul(x, 16777619); } return x >>> 0; };

const ALL = dv.pages('"04-题库" or "05-真题库"').where(p =>
  (p.type === "题目" || p.type === "真题") && p.status !== "deprecated" &&
  p.teaching_level === "拓展" &&
  (!EXCLUDE_USED || !hasUsed(p.used_in))).array();

// ── KP 候选清单（档内未用题 KP 频次 top 20，帮用户选准 KP 名）──
const kpCnt = new Map();
for (const p of ALL) for (const k of new Set(kpNames(p))) kpCnt.set(k, (kpCnt.get(k) || 0) + 1);
dv.header(4, `📖 习题书卷候选：${ALL.filter(p => KPS.some(k => kpNames(p).includes(k))).length} 题（档内总量 ${ALL.length}）`);
dv.paragraph("**可填 KP**（拓展档未用题 top 20）：" +
  [...kpCnt.entries()].sort((a, b) => b[1] - a[1]).slice(0, 20)
    .map(([k, v]) => `${k}×${v}`).join(" ｜ "));

// ── 成卷：KP 精确命中 → 难度升序 ──
const pool = ALL.filter(p => KPS.some(k => kpNames(p).includes(k)));
pool.sort((a, b) => diffNum(a.difficulty) - diffNum(b.difficulty) ||
                     h(SEED + ":" + a.file.path) - h(SEED + ":" + b.file.path));
const picked = pool.slice(0, Math.min(TOTAL, pool.length));

dv.table(["#", "题目", "难度", "题型", "模块", "来源"],
  picked.map((p, i) => [
    i + 1, p.file.link, p.difficulty,
    (Array.isArray(p.question_type) ? p.question_type.join("/") : p.question_type) || "—",
    p.subject_module, String(p.source || "").slice(0, 20)
  ]));
const kpCover = KPS.map(k => `${k}×${picked.filter(p => kpNames(p).includes(k)).length}`).join(" ｜ ");
dv.paragraph(`**KP 覆盖**：${kpCover}`);
dv.paragraph("**参考答案**（卷尾）\n：\n" +
  picked.map((p, i) => `${i + 1}. ${p.file.link} → 见题目笔记「参考答案」`).join("\n"));
dv.paragraph("**复制清单**（粘到试卷文件后跑 mark_used.py 回填）\n：" +
  picked.map((p, i) => `${i + 1}. ${p.file.link}`).join("\n"));
```

## 预设 3 · 测试题卷（竞赛档 · 整卷模考）

```dataviewjs
// ══════════ 预设 3 · 测试题卷 ══════════
// 竞赛档整卷模考：2:5:3 梯度 + 限时 120min + 同来源限流 + 独立答案卷。
const LEVEL  = "竞赛";      // 测试题档（三模块视图，勿改）
const STAGE  = "";          // 初赛 / 决赛 / "" 不限
const TOTAL  = 50;          // 整卷模考 50 题
const TIME_MIN = 120;       // 限时 120 分钟
const GRADIENT = [0.2, 0.5, 0.3];   // 易(d≤2) : 中(d=3) : 难(d≥4)，即 2:5:3
const QUOTA  = {};          // 模块配额上限，如 {化学原理:15}；{} 不限
const MAX_PER_SOURCE = 3;   // 同来源限流
const SEED   = 20260902;

// ── 工具函数（dataviewjs 块间不共享作用域，每块重定义）──
const hasUsed = u => u != null && (Array.isArray(u) ? u.length > 0 : true);
const diffNum = d => { const m = String(d ?? "").match(/\d+/); return m ? Number(m[0]) : NaN; };
const stageHit = (v, want) => String(v ?? "").split("/").some(x => x.trim() === want);
function srcKey(s) {
  let k = String(s || "").trim();
  k = k.replace(/[（(]忠实转录[)）]/g, "");
  k = k.replace(/[·•]\s*第\s*\d+\s*[讲章][\s\S]*$/, "");
  k = k.replace(/第\s*[\d一二三四五六七八九十]+\s*[章节讲篇][\s\S]*$/, "");
  k = k.replace(/[·•]\s*第\s*[\d一二三四五六七八九十]+\s*[分册卷][\s\S]*$/, "");
  k = k.replace(/-\s*\d{1,2}\s*$/, "");
  k = k.replace(/[\s·•、,，\-—]+$/, "");
  return k.trim() || "(未知来源)";
}
function mkRand(seed) {
  let a = seed >>> 0;
  return function () {
    a = (a + 0x6D2B79F5) >>> 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
const rand = mkRand(SEED);

const pool = dv.pages('"04-题库" or "05-真题库"').where(p => {
  if (p.type !== "题目" && p.type !== "真题") return false;
  if (p.status === "deprecated") return false;
  if (p.teaching_level !== LEVEL) return false;
  if (STAGE && !stageHit(p.exam_stage, STAGE)) return false;
  if (hasUsed(p.used_in)) return false;
  const d = diffNum(p.difficulty);
  return d >= 1 && d <= 5;
}).array();

// ── 分档与名额（沿用智能组卷器：余数补给候选最多的档）──
const bands = [
  { name: "易", list: pool.filter(p => diffNum(p.difficulty) <= 2) },
  { name: "中", list: pool.filter(p => diffNum(p.difficulty) === 3) },
  { name: "难", list: pool.filter(p => diffNum(p.difficulty) >= 4) },
];
let quotaN = GRADIENT.map(g => Math.floor(TOTAL * g));
let rest = TOTAL - quotaN.reduce((a, b) => a + b, 0);
const bySupply = bands.map((b, i) => i).sort((i, j) => bands[j].list.length - bands[i].list.length);
for (let k = 0; rest > 0; k++, rest--) quotaN[bySupply[k % 3]]++;
bands.forEach((b, i) => b.target = quotaN[i]);

// ── 抽取（同来源限流 + 模块配额）──
const usedSrc = new Map(), usedMod = new Map();
function pick(band) {
  const rest = band.list.slice();
  const out = [];
  while (out.length < band.target && rest.length) {
    const i = Math.floor(rand() * rest.length);
    const p = rest.splice(i, 1)[0];
    const k = srcKey(p.source);
    const m = p.subject_module || "(未标模块)";
    if (MAX_PER_SOURCE > 0 && (usedSrc.get(k) || 0) >= MAX_PER_SOURCE) continue;
    if (QUOTA[m] !== undefined && (usedMod.get(m) || 0) >= QUOTA[m]) continue;
    usedSrc.set(k, (usedSrc.get(k) || 0) + 1);
    usedMod.set(m, (usedMod.get(m) || 0) + 1);
    out.push(p);
  }
  return out;
}
bands.sort((a, b) => (a.list.length - a.target) - (b.list.length - b.target));
const picked = [];
bands.forEach(b => { b.got = pick(b); picked.push(...b.got); });
picked.sort((a, b) => diffNum(a.difficulty) - diffNum(b.difficulty));

// ── 输出：卷面 + 分档诊断 + 独立答案卷 ──
const avg = picked.reduce((s, p) => s + diffNum(p.difficulty), 0) / (picked.length || 1);
dv.header(4, `⏱️ 模考卷 ${picked.length}/${TOTAL} 题 ｜ 限时 ${TIME_MIN} 分钟 ｜ 均难度 ${avg.toFixed(2)} ｜ 种子 ${SEED}`);
dv.table(["#", "题目", "难度", "题型", "模块", "来源(归一)"],
  picked.map((p, i) => [
    i + 1, p.file.link, p.difficulty,
    (Array.isArray(p.question_type) ? p.question_type.join("/") : p.question_type) || "—",
    p.subject_module, srcKey(p.source).slice(0, 18)
  ]));
const diag = bands.map(b =>
  `  ${b.name}档：目标 ${b.target}／实得 ${b.got.length}（候选 ${b.list.length}）` +
  (b.got.length < b.target ? " ⚠️ 候选不足" : ""));
dv.paragraph("**分档诊断**\n" + diag.join("\n"));
dv.paragraph("**独立答案卷**（整段复制到「<卷名>-答案.md」，与卷面分开）\n：\n" +
  picked.map((p, i) => `${i + 1}. ${p.file.link} → 见题目笔记「参考答案」`).join("\n"));
dv.paragraph("**复制清单**（粘到试卷文件后跑 mark_used.py 回填）\n：" +
  picked.map((p, i) => `${i + 1}. ${p.file.link}`).join("\n"));
```
