$$
\begin{array}{r l} & {\mathrm {O_ {2} + e^ {-} + Li^ {+} = LiO_ {2} (1分)}} \\ & {2 \mathrm {LiO_ {2} = Li_ {2} O_ {2} + O_ {2} (1分)}} \\ & {\mathrm {LiO_ {2} + e^ {-} + Li^ {+} = Li_ {2} O_ {2} (1分)}} \\ & {\mathrm {Li_ {2} O_ {2} + 2e^ {-} + 2Li^ {+} = 2Li_ {2} O(1分)}} \end{array}
$$
