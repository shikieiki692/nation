$$

$$
\mathrm{CH} _ {4} (\mathrm{g}) + \mathrm{H} _ {2} \mathrm{O} (\mathrm{g}) \rightarrow \mathrm{CO} (\mathrm{g}) + 3 \mathrm{H} _ {2} (\mathrm{g})
