$$
\mathrm{I} \xrightarrow {   } \mathrm{Cl} + \mathrm{Ph} _{2} \mathrm{S} \xrightarrow {\mathrm{AgBF} _{4} (1  eq. )} \mathrm{A} \xrightarrow {\mathrm{NaH}} [ \mathrm{B} ]
$$
