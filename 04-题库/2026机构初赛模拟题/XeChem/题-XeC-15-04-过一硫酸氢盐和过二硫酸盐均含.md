$$
\begin{array}{r l} n (\mathrm {KIO_ {3}}) & = \frac {m (\mathrm {KI O_ {3}})}{M (\mathrm {KI O_ {3}})} \times \frac {V _ {\text {移取}}}{V _ {\text {总}}} \\ & = \frac {0.3566 \mathrm{g}}{214.00 \mathrm{g/mol}} \times \frac {25.00 \mathrm{mL}}{250.00 \mathrm{mL}} \\ & = 1.6664 \times 10 ^ {- 4} \mathrm{mol} \end{array}
$$

$$
\begin{array}{r l} c (\mathrm{Na} _ {2} \mathrm{S} _ {2} \mathrm{O} _ {3}) & = \frac {n (\mathrm{Na} _ {2} \mathrm{S} _ {2} \mathrm{O} _ {3})}{V (\mathrm{Na} _ {2} \mathrm{S} _ {2} \mathrm{O} _ {3})} \\ & = \frac {6 \times n (\mathrm{KIO} _ {3})}{V (\mathrm{Na} _ {2} \mathrm{S} _ {2} \mathrm{O} _ {3})} \\ & = \frac {6 \times 1.6664 \times 10 ^ {- 4} \mathrm{mol}}{0.02005 \mathrm{L}} \\ & = 0.04987 \mathrm{mol/L} \end{array}
